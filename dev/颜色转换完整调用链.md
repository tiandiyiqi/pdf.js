getRgbHex(src, srcOffset) {
  // 处理CMYK颜色空间（4个组件）
  if (this.numComps === 4) {
    // 创建过滤后的数据副本
    const filteredSrc = new Float32Array(4);
    filteredSrc[0] = src[srcOffset]; // C分量
    filteredSrc[1] = src[srcOffset + 1]; // M分量
    filteredSrc[2] = src[srcOffset + 2]; // Y分量
    filteredSrc[3] = src[srcOffset + 3]; // K分量

    // 应用ColorConverter的CMYK过滤算法
    const filtered = ColorConverter.filterCMYK([
      filteredSrc[0],
      filteredSrc[1],
      filteredSrc[2],
      filteredSrc[3],
    ]);

    // 将过滤后的值复制回临时数组
    filteredSrc[0] = filtered[0];
    filteredSrc[1] = filtered[1];
    filteredSrc[2] = filtered[2];
    filteredSrc[3] = filtered[3];

    // 使用过滤后的数据进行转换
    this.#convertPixel(filteredSrc, 0, /* css */ true);
  } else {
    // 非CMYK颜色空间（如RGB或灰度）：直接使用原始数据转换
    this.#convertPixel(src, srcOffset, /* css */ true);
  }
  return QCMS._cssColor;
}

// 文件：src/core/colorspace.js
// 行号：878-973

#toRgb(src, srcOffset, srcScale, dest, destOffset, skipFilter = false) {
  let c = src[srcOffset] * srcScale;
  let m = src[srcOffset + 1] * srcScale;
  let y = src[srcOffset + 2] * srcScale;
  let k = src[srcOffset + 3] * srcScale;

  // Apply color filter (unless skipFilter is true)
  if (!skipFilter) {
    const filteredCMYK = ColorConverter.filterCMYK([c, m, y, k]);
    c = filteredCMYK[0];
    m = filteredCMYK[1];
    y = filteredCMYK[2];
    k = filteredCMYK[3];
  }

  // 使用多项式回归算法计算 RGB
  dest[destOffset] = 255 + c * (-4.387332384609988 * c + ...) + ...;
  dest[destOffset + 1] = 255 + c * (8.841041422036149 * c + ...) + ...;
  dest[destOffset + 2] = 255 + c * (0.8842522430003296 * c + ...) + ...;
}

```javascript
// 文件：src/core/icc_colorspace.js
// 行号：206-275

getRgbBuffer(src, srcOffset, count, dest, destOffset, bits, alpha01) {
  // 1. 准备数据：截取需要处理的数据段
  src = src.subarray(srcOffset, srcOffset + count * this.numComps);

  // 2. 位深转换：如果不是8位数据，进行归一化处理
  if (bits !== 8) {
    const scale = 255 / ((1 << bits) - 1);
    for (let i = 0, ii = src.length; i < ii; i++) {
      src[i] *= scale;
    }
  }

  // 3. CMYK 特殊处理
  if (this.numComps === 4) {
    // 创建过滤后的数据副本
    const filteredSrc = new Float32Array(src.length);
    for (let i = 0; i < count; i++) {
      const offset = i * 4;
      // 应用 ColorConverter 的 CMYK 过滤
      const filtered = ColorConverter.filterCMYK([
        src[offset],
        src[offset + 1],
        src[offset + 2],
        src[offset + 3],
      ]);
      // 存储过滤后的值
      filteredSrc[offset] = filtered[0];
      filteredSrc[offset + 1] = filtered[1];
      filteredSrc[offset + 2] = filtered[2];
      filteredSrc[offset + 3] = filtered[3];
    }

    // 执行实际的颜色转换（使用过滤后的数据）
    QCMS._mustAddAlpha = alpha01 && dest.buffer === filteredSrc.buffer;
    QCMS._destBuffer = dest;
    QCMS._destOffset = destOffset;
    QCMS._destLength = count * (3 + alpha01);
    qcms_convert_array(this.#transformer, filteredSrc);
    // 清理状态
    QCMS._mustAddAlpha = false;
    QCMS._destBuffer = null;
  } else {
    // 非 CMYK 颜色空间：直接转换原始数据
    // ...
  }
}
```

```javascript
// 文件：src/core/colorspace.js
// 行号：467-521

getRgbItem(src, srcOffset, dest, destOffset) {
  if (typeof PDFJSDev === "undefined" || PDFJSDev.test("TESTING")) {
    assert(
      dest instanceof Uint8ClampedArray,
      'AlternateCS.getRgbItem: Unsupported "dest" type.'
    );
  }

  // 检查专色可见性（二元过滤：全有或全无）
  const cmykNames = ["Cyan", "Magenta", "Yellow", "Black"];
  if (this.channelNames && this.channelNames.length > 0) {
    // 检查是否有专色通道（非CMYK通道）
    const spotChannels = this.channelNames.filter(
      name => !cmykNames.includes(name)
    );

    if (spotChannels.length > 0) {
      // 对于单通道专色（Separation），检查第一个通道
      // 对于多通道（DeviceN），检查所有专色通道
      let shouldShow = true;
      for (let i = 0; i < this.channelNames.length; i++) {
        const channelName = this.channelNames[i];
        if (!cmykNames.includes(channelName)) {
          // 这是专色通道
          const spotValue = src[srcOffset + i];
          // 如果专色值 > 0，检查可见性
          if (spotValue > 0) {
            const filteredValue = ColorConverter.filterSpot(
              channelName,
              spotValue
            );
            if (filteredValue === 0) {
              // 专色不可见，返回白色
              shouldShow = false;
              break;
            }
          }
        }
      }

      if (!shouldShow) {
        // 专色不可见，直接返回白色RGB
        dest[destOffset] = 255;
        dest[destOffset + 1] = 255;
        dest[destOffset + 2] = 255;
        return;
      }
    }
  }

  // 专色可见或没有专色，继续原有的转换流程
  const tmpBuf = this.tmpBuf;
  this.tintFn(src, srcOffset, tmpBuf, 0);  // ← 将专色转换为 CMYK
  this.base.getRgbItem(tmpBuf, 0, dest, destOffset, true); // ← skipFilter = true，跳过颜色过滤
}
```

**3. AlternateCS.getRgbBuffer()（批量专色转换）**

```javascript
// 文件：src/core/colorspace.js
// 行号：523-626

getRgbBuffer(src, srcOffset, count, dest, destOffset, bits, alpha01) {
  // ... 参数验证 ...

  const tintFn = this.tintFn;
  const base = this.base;
  const scale = 1 / ((1 << bits) - 1);
  const baseNumComps = base.numComps;
  const usesZeroToOneRange = base.usesZeroToOneRange;
  const isPassthrough =
    (base.isPassthrough(8) || !usesZeroToOneRange) && alpha01 === 0;
  let pos = isPassthrough ? destOffset : 0;
  const baseBuf = isPassthrough
    ? dest
    : new Uint8ClampedArray(baseNumComps * count);
  const numComps = this.numComps;

  // 检查是否有专色通道（用于二元过滤）
  const cmykNames = ["Cyan", "Magenta", "Yellow", "Black"];
  const hasSpotColors =
    this.channelNames &&
    this.channelNames.length > 0 &&
    this.channelNames.some(name => !cmykNames.includes(name));

  const scaled = new Float32Array(numComps);
  const tinted = new Float32Array(baseNumComps);
  let i, j;
  let currentSrcOffset = srcOffset; // 保存当前源数据偏移量

  for (i = 0; i < count; i++) {
    // 读取当前像素的所有分量
    for (j = 0; j < numComps; j++) {
      scaled[j] = src[currentSrcOffset + j] * scale;
    }

    // 检查专色可见性（二元过滤：全有或全无）
    let shouldShow = true;
    if (hasSpotColors) {
      for (j = 0; j < this.channelNames.length; j++) {
        const channelName = this.channelNames[j];
        if (!cmykNames.includes(channelName)) {
          // 这是专色通道
          const spotValue = scaled[j]; // 已经经过scale转换
          // 如果专色值 > 0，检查可见性
          if (spotValue > 0) {
            const filteredValue = ColorConverter.filterSpot(
              channelName,
              spotValue
            );
            if (filteredValue === 0) {
              // 专色不可见，填充白色
              shouldShow = false;
              break;
            }
          }
        }
      }
    }

    if (!shouldShow) {
      // 专色不可见，直接填充白色RGB
      if (isPassthrough) {
        // 直接写入dest
        dest[destOffset + i * (3 + alpha01)] = 255;
        dest[destOffset + i * (3 + alpha01) + 1] = 255;
        dest[destOffset + i * (3 + alpha01) + 2] = 255;
        if (alpha01 === 1) {
          dest[destOffset + i * (3 + alpha01) + 3] = 255;
        }
      } else {
        // 写入baseBuf（后续会通过base.getRgbBuffer转换）
        // 对于白色，baseBuf应该是[255, 255, 255]（如果base是RGB）
        // 但这里base可能是CMYK，所以我们需要填充白色对应的base颜色空间值
        // 最简单的方法是填充0（对于CMYK，白色是[0,0,0,0]）
        for (j = 0; j < baseNumComps; j++) {
          baseBuf[pos + j] = 0;
        }
        pos += baseNumComps;
      }
      currentSrcOffset += numComps; // 移动到下一个像素
      continue;
    }

    // 专色可见或没有专色，继续原有的转换流程
    tintFn(scaled, 0, tinted, 0);
    if (usesZeroToOneRange) {
      for (j = 0; j < baseNumComps; j++) {
        baseBuf[pos++] = tinted[j] * 255;
      }
    } else {
      base.getRgbItem(tinted, 0, baseBuf, pos, true); // ← skipFilter = true
      pos += baseNumComps;
    }
    currentSrcOffset += numComps; // 移动到下一个像素
  }

  if (!isPassthrough) {
    base.getRgbBuffer(baseBuf, 0, count, dest, destOffset, 8, alpha01);
  }
}
```

## 三、颜色过滤核心逻辑

### 3.1 CMYK 通道过滤

**文件：** `src/core/color_converter.js`  
**方法：** `ColorConverter.filterCMYK()`

```javascript
// 文件：src/core/color_converter.js
// 行号：187-240

/**
 * 过滤CMYK颜色值，根据当前颜色过滤器配置隐藏/显示特定颜色通道
 * @param {number[]} cmyk - 输入的CMYK颜色值数组，顺序为[C, M, Y, K]
 * @returns {number[]} 过滤后的CMYK颜色值数组
 */
static filterCMYK(cmyk) {
  // 如果过滤器未启用，直接返回原始CMYK值
  if (!this.#colorFilterConfig.enabled) {
    return [...cmyk];
  }

  // 创建CMYK数组的副本用于过滤操作
  const filtered = [...cmyk];
  const colors = this.#colorFilterConfig.colors;

  // 根据配置逐个检查颜色通道是否应该显示
  // 如果颜色通道被禁用(visible=false)，则将该通道值设为0
  if (!colors.get("Cyan")) {
    filtered[0] = 0; // 青色通道
  }
  if (!colors.get("Magenta")) {
    filtered[1] = 0; // 品红色通道
  }
  if (!colors.get("Yellow")) {
    filtered[2] = 0; // 黄色通道
  }
  if (!colors.get("Black")) {
    filtered[3] = 0; // 黑色通道
  }

  return filtered;
}
```

### 3.2 专色二元过滤

**文件：** `src/core/color_converter.js`  
**方法：** `ColorConverter.filterSpot()`

```javascript
// 文件：src/core/color_converter.js
// 行号：242-260

/**
 * 过滤专色值（二元过滤：全有或全无）
 * @param {string} spotName - 专色名称
 * @param {number} spotValue - 专色值
 * @returns {number} 过滤后的专色值（0 表示不可见，原值表示可见）
 */
static filterSpot(spotName, spotValue) {
  if (!this.#colorFilterConfig.enabled) {
    return spotValue;
  }

  const visible = this.#colorFilterConfig.colors.get(spotName);
  if (visible === false) {
    return 0; // 专色不可见，返回 0
  }

  return spotValue; // 专色可见，返回原值
}
```

### 4.2 大尺寸图像的自动降采样机制

当PDF中包含超大尺寸的图像时，PDF.js会自动进行**降采样（downsampling）**处理，将图像降低到合适的分辨率以避免内存溢出和性能问题。这就是为什么你看到的是低分辨率图像而不是原始高分辨率图像。

#### 重要说明

**这不是"链接丢失"的回退机制**，而是PDF.js针对所有超大图像（包括嵌入图像）的性能优化策略。即使图像完整嵌入在PDF中，只要超过特定阈值，都会被降采样处理。

**2. 降采样倍数计算算法** (`src/core/image_resizer.js:99-121`)

```javascript
static getReducePowerForJPX(width, height, componentsCount) {
  const area = width * height;

  // WASM运行时最大2GB内存，每个颜色通道4字节
  // 限制总内存为 1GB / (componentsCount * 4)
  const maxJPXArea = 2 ** 30 / (componentsCount * 4);

  // 先检查是否超过canvas最大尺寸
  if (!this.needsToBeResized(width, height)) {
    if (area > maxJPXArea) {
      // 仅超过JPX内存限制，计算需要的缩小倍数
      // 例如：area = 4 * maxJPXArea，则返回 ceil(log2(4)) = 2
      return Math.ceil(Math.log2(area / maxJPXArea));
    }
    return 0;  // 不需要降采样
  }

  // 超过canvas最大尺寸和JPX内存限制
  const { MAX_DIM, MAX_AREA } = this;
  const minFactor = Math.max(
    width / MAX_DIM,     // 宽度超限的缩放因子
    height / MAX_DIM,    // 高度超限的缩放因子
    Math.sqrt(area / Math.min(maxJPXArea, MAX_AREA))  // 面积超限的缩放因子
  );

  // 返回大于minFactor的最小2的幂次方
  // 例如：minFactor=3.5，返回 ceil(log2(3.5))=2，实际缩放factor=2^2=4
  return Math.ceil(Math.log2(minFactor));
}
```

**3. 通用图像降采样判断** (`src/core/image_resizer.js:56-97`)

```javascript
static needsToBeResized(width, height) {
  // MIN_IMAGE_DIM = 2048
  if (width <= this.#goodSquareLength && height <= this.#goodSquareLength) {
    return false;  // 图像够小，不需要降采样
  }

  const { MAX_DIM } = this;  // MAX_DIM 约为 65537
  if (width > MAX_DIM || height > MAX_DIM) {
    return true;  // 任一维度超过最大值，必须降采样
  }

  const area = width * height;
  if (this._hasMaxArea) {
    return area > this.MAX_AREA;  // 面积超过最大值
  }

  if (area < this.#goodSquareLength ** 2) {
    return false;  // 面积足够小
  }

  // 动态计算MAX_AREA（基于canvas实际能分配的最大内存）
  // 并更新 #goodSquareLength
  if (this._areGoodDims(width, height)) {
    this.#goodSquareLength = Math.max(
      this.#goodSquareLength,
      Math.floor(Math.sqrt(width * height))
    );
    return false;
  }

  // 二分查找canvas能支持的最大尺寸
  this.#goodSquareLength = this._guessMax(
    this.#goodSquareLength,
    MAX_DIM,
    MAX_ERROR,
    0
  );
  const maxArea = (this.MAX_AREA = this.#goodSquareLength ** 2);

  return area > maxArea;
}
```

#### 降采样触发条件：

**1. 尺寸限制**（三个条件之一满足即触发）：

- **单维度超限**：`width > 65537` 或 `height > 65537`
- **面积超限**：`width × height > MAX_AREA`（动态计算，约为 2048² 到 65537²）
- **JPX内存超限**：`width × height > 2³⁰ / (componentsCount × 4)`

**2. 降采样倍数计算公式**：

```
reducePower = Math.ceil(Math.log2(缩放因子))
实际缩放倍数 = 2^reducePower

示例：
- 原图 8192×8192，MAX_DIM=4096
- 缩放因子 = max(8192/4096, 8192/4096) = 2
- reducePower = ceil(log2(2)) = 1
- 实际倍数 = 2^1 = 2
- 结果尺寸 = 4096×4096
```

**3. 降采样算法特点**：

- **只能是2的幂次方**：缩放倍数必须是 2, 4, 8, 16, 32... 不能是任意值
- **向上取整**：如果计算出缩放因子是3，会使用4（2²）而不是2
- **保证不超限**：降采样后的尺寸一定满足所有限制条件

#### 重要说明：

1. **不是链接丢失**：你看到的低分辨率图像**不是**因为链接图像找不到，而是PDF.js主动降采样的结果。即使是完整嵌入的图像，只要超过阈值就会被降采样。

2. **性能优化**：降采样是为了：
   - 避免浏览器内存溢出
   - 加快解码和渲染速度
   - 支持在低配设备上查看大尺寸PDF

3. **JPX特殊处理**：JPX/JPEG2000格式支持在解码时直接降采样，比先解码再缩放效率高得多，所以有专门的`getReducePowerForJPX`方法。

4. **可配置性**：通过修改`canvasMaxAreaInBytes`可以允许更大的图像，但需要权衡内存和性能。

5. **OPI链接图像**：如果PDF使用OPI（Open Prepress Interface）规范，可能包含：
   - 低分辨率预览图（嵌入在PDF中）
   - 高分辨率图像链接（外部文件路径）

   PDF.js会渲染嵌入的低分辨率预览图，但**不会**尝试加载外部高分辨率图像。这是设计决策，不是bug。

6. **查看原始尺寸**：如果想知道图像被降采样了多少，可以在控制台查看警告信息：
   ```
   PDFImage - using the Width/Height of the image data, rather than the image dictionary.
   ```
   或者检查`reducePower`值（如果是JPX图像）。

#### 重要结论：

**缩略图回退机制的特点**：

1. **独立于颜色转换调用链**：缩略图生成使用Canvas 2D上下文直接渲染，不经过完整的颜色转换流程
2. **基于已渲染页面内容**：缩略图基于成功渲染的页面内容生成，包括文本、图形和可用的嵌入图像
3. **提供内容概览**：即使链接图像缺失，用户仍能看到页面内容的预览
4. **视觉指示**：通过CSS类`missingThumbnailImage`标识缩略图状态

### 4.3 关键设计要点

#### 4.3.1 skipFilter 参数

为了避免专色的 CMYK 替代色被错误地按通道过滤，引入了 `skipFilter` 参数：

- **DeviceCmykCS.getRgbItem()** 和 **DeviceCmykCS.#toRgb()** 支持 `skipFilter` 参数
- **IccColorSpace.getRgbItem()** 支持 `skipFilter` 参数
- **AlternateCS.getRgbItem()** 在调用 `base.getRgbItem()` 时传递 `skipFilter = true`

这样可以确保：

- **CMYK 基础色**：按通道过滤（Cyan=false 时，C 通道设为 0）
- **专色**：二元过滤（专色=false 时，整个专色显示为白色，其 CMYK 替代色不被过滤）

## 五、总结

### 5.1 CMYK 颜色转换路径

1. **矢量图形/文字填色：**
   - ICC-based: `evaluator.js` → `IccColorSpace.getRgbHex()` → `ColorConverter.filterCMYK()` → `#convertPixel()`
   - DeviceCMYK: `evaluator.js` → `DeviceCmykCS.getRgbHex()` → `getRgb()` → `getRgbItem()` → `#toRgb()` → `ColorConverter.filterCMYK()`

2. **图像颜色转换：**
   - ICC-based: `image.js` → `fillRgb()` → `IccColorSpace.getRgbBuffer()` → `ColorConverter.filterCMYK()` → `qcms_convert_array()`
   - DeviceCMYK: `image.js` → `fillRgb()` → `DeviceCmykCS.getRgbBuffer()` → 循环 `#toRgb()` → `ColorConverter.filterCMYK()`

### 5.2 专色转换路径

1. **矢量图形/文字填色：**
   - `evaluator.js` → `AlternateCS.getRgbHex()` → `getRgb()` → `getRgbItem()` → `ColorConverter.filterSpot()` → `tintFn()` → `base.getRgbItem(skipFilter=true)`

2. **图像颜色转换：**
   - `image.js` → `fillRgb()` → `AlternateCS.getRgbBuffer()` → `ColorConverter.filterSpot()` → `tintFn()` → `base.getRgbBuffer()`

### 5.3 颜色过滤规则

- **CMYK 基础色**：通道级过滤（每个通道独立控制）
- **专色**：二元过滤（全有或全无，不可见时显示为白色）
