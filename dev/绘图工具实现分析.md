#### InkDrawOutliner（绘图过程管理器）

**初始化：**

```javascript
    [x, y] = this.#normalizePoint(x, y);

    const line = (this.#line = [NaN, NaN, NaN, NaN, x, y]);
    this.#points = [x, y];
```

**添加点（绘图过程）：**

```javascript
add(x, y) {
  [x, y] = this.#normalizePoint(x, y);
  const [x1, y1, x2, y2] = this.#last.subarray(2, 6);
  const diffX = x - x2;
  const diffY = y - y2;
  const d = Math.hypot(this.#parentWidth * diffX, this.#parentHeight * diffY);

  if (d <= 2) {
    // 避免垃圾点，点太接近时忽略
    return null;
  }

  this.#points.push(x, y);

  if (isNaN(x1)) {
    // 只有一个点时的处理
    this.#last.set([x2, y2, x, y], 2);
    this.#line.push(NaN, NaN, NaN, NaN, x, y);
    return {
      path: {
        d: this.toSVGPath(),
      },
    };
  }

  // 使用贝塞尔曲线平滑连接点
  this.#last.set([x1, y1, x2, y2, x, y], 0);
  this.#line.push(...Outline.createBezierPoints(x1, y1, x2, y2, x, y));

  return {
    path: {
      d: this.toSVGPath(),
    },
  };
}
```

**生成 SVG 路径：**

```javascript
toSVGPath() {
  const firstX = Outline.svgRound(this.#line[4]);
  const firstY = Outline.svgRound(this.#line[5]);

  if (this.#points.length === 2) {
    // 单点：绘制一个点
    this.#lastSVGPath = `${this.#lastSVGPath} M ${firstX} ${firstY} Z`;
    return this.#lastSVGPath;
  }

  if (this.#points.length === 4) {
    // 两点：绘制直线
    const secondX = Outline.svgRound(this.#line[10]);
    const secondY = Outline.svgRound(this.#line[11]);
    this.#lastSVGPath = `${this.#lastSVGPath} L ${secondX} ${secondY}`;
    return this.#lastSVGPath;
  }

  // 多点：使用贝塞尔曲线
  const buffer = [];
  for (let i = this.#lastIndex, ii = this.#line.length; i < ii; i += 6) {
    const [c1x, c1y, c2x, c2y, x, y] = this.#line
      .slice(i, i + 6)
      .map(Outline.svgRound);
    buffer.push(`C${c1x} ${c1y} ${c2x} ${c2y} ${x} ${y}`);
  }
  this.#lastSVGPath += buffer.join(" ");

  return this.#lastSVGPath;
}
```

```javascript
startDrawingSession(event) {
  this.div.focus({
    preventScroll: true,

  if (this.#drawingAC) {

    ({ relatedTarget }) => {

        this.#focusedElement = null;
        this.commitOrRemove();
      }
    },
    { signal }
  );

}
```

### 3.2 绘图事件处理

**文件位置：** `src/display/editor/draw.js` `startDrawing` 方法（第673-795行）

**事件监听设置：**

```javascript
static startDrawing(parent, uiManager, _isLTR, event) {
  const { target, offsetX: x, offsetY: y, pointerId, pointerType } = event;

  // 检查指针类型
  if (CurrentPointers.isInitializedAndDifferentPointerType(pointerType)) {
    return;
  }

  const ac = (DrawingEditor.#currentDrawingAC = new AbortController());
  const signal = parent.combinedSignal(ac);

  CurrentPointers.setPointer(pointerType, pointerId);

  // 监听 pointerup 事件（结束绘图）
  window.addEventListener(
    "pointerup",
    e => {
      if (CurrentPointers.isSamePointerIdOrRemove(e.pointerId)) {
        this._endDraw(e);
      }
    },
    { signal }
  );

  // 监听 pointermove 事件（绘图中）
  target.addEventListener("pointermove", this._drawMove.bind(this), {
    signal,
  });

  // 创建绘图器实例
  DrawingEditor.#currentDraw = this.createDrawerInstance(
    x,
    y,
    parentWidth,
    parentHeight,
    rotation
  );

  // 创建绘图选项
  DrawingEditor.#currentDrawingOptions = this.getDefaultDrawingOptions();
  this._currentParent = parent;

  // 在 DrawLayer 中创建 SVG 元素
  ({ id: this._currentDrawId } = parent.drawLayer.draw(
    this._mergeSVGProperties(
      DrawingEditor.#currentDrawingOptions.toSVGProperties(),
      DrawingEditor.#currentDraw.defaultSVGProperties
    ),
    /* isPathUpdatable = */ true,
    /* hasClip = */ false
  ));
}
```

**移动处理：**

```javascript
static _drawMove(event) {
  if (!DrawingEditor.#currentDraw) {
    return;
  }

  const { offsetX, offsetY, pointerId } = event;

  if (!CurrentPointers.isSamePointerId(pointerId)) {
    return;
  }

  // 更新 DrawLayer 中的路径
  this._currentParent.drawLayer.updateProperties(
    this._currentDrawId,
    DrawingEditor.#currentDraw.add(offsetX, offsetY)
  );

  stopEvent(event);
}
```

**结束绘图：**

```javascript
static _endDraw(event) {
  const parent = this._currentParent;
  if (!parent) {
    return;
  }

  parent.toggleDrawing(true);
  this._cleanup(false);

  if (event?.target === parent.div) {
    parent.drawLayer.updateProperties(
      this._currentDrawId,
      DrawingEditor.#currentDraw.end(event.offsetX, event.offsetY)
    );
  }

  this.endDrawing(/* isAborted = */ false);
}

static endDrawing(isAborted) {
  const parent = this._currentParent;
  if (!parent) {
    return null;
  }

  if (!DrawingEditor.#currentDraw.isEmpty()) {
    const {
      pageDimensions: [pageWidth, pageHeight],
      scale,
    } = parent;

    // 创建编辑器实例
    const editor = parent.createAndAddNewEditor(
      { offsetX: 0, offsetY: 0 },
      false,
      {
        drawId: this._currentDrawId,
        drawOutlines: DrawingEditor.#currentDraw.getOutlines(
          pageWidth * scale,
          pageHeight * scale,
          scale,
          this._INNER_MARGIN
        ),
        drawingOptions: DrawingEditor.#currentDrawingOptions,
        mustBeCommitted: !isAborted,
      }
    );

    this._cleanup(true);
    return editor;
  }

  parent.drawLayer.remove(this._currentDrawId);
  this._cleanup(true);
  return null;
}
```

## 4. 前端渲染机制

### 4.1 DrawLayer（绘图层）

**文件位置：** `src/display/draw_layer.js`

**核心功能：** 管理 PDF 页面上的 SVG 绘图元素

**创建 SVG 元素：**

```javascript
draw(properties, isPathUpdatable = false, hasClip = false) {
  const id = DrawLayer.#id++;
  const root = this.#createSVG();  // 创建 SVG 根元素

  const defs = DrawLayer._svgFactory.createElement("defs");
  root.append(defs);

  const path = DrawLayer._svgFactory.createElement("path");
  defs.append(path);

  const pathId = `path_p${this.pageIndex}_${id}`;
  path.setAttribute("id", pathId);
  path.setAttribute("vector-effect", "non-scaling-stroke");

  if (isPathUpdatable) {
    this.#toUpdate.set(id, path);
  }

  const use = DrawLayer._svgFactory.createElement("use");
  root.append(use);
  use.setAttribute("href", `#${pathId}`);

  this.updateProperties(root, properties);
  this.#mapping.set(id, root);

  return { id, clipPathId: `url(#${clipPathId})` };
}
```

**更新属性：**

```javascript
updateProperties(elementOrId, properties) {
  if (!properties) {
    return;
  }

  const { root, bbox, rootClass, path } = properties;
  const element =
    typeof elementOrId === "number"
      ? this.#mapping.get(elementOrId)
      : elementOrId;

  if (!element) {
    return;
  }

  if (root) {
    this.#updateProperties(element, root);
  }

  if (bbox) {
    DrawLayer.#setBox(element, bbox);  // 设置位置和大小
  }

  if (rootClass) {
    const { classList } = element;
    for (const [className, value] of Object.entries(rootClass)) {
      classList.toggle(className, value);
    }
  }

  if (path) {
    const defs = element.firstElementChild;
    const pathElement = defs.firstElementChild;
    this.#updateProperties(pathElement, path);  // 更新路径数据
  }
}
```

### 5.2 与 PDF 批注系统集成

```javascript
class InkAnnotationElement extends AnnotationElement {
  constructor(parameters) {
    super(parameters, { isRenderable: true, ignoreBorder: true });
    this.containerClassName = "inkAnnotation";
    this.svgElementName = "svg:polyline";
    this.annotationEditorType = AnnotationEditorType.INK;
  }

  render() {
    // 渲染 SVG polyline 元素
    const svg = this.svgFactory.create(width, height);
    const g = this.svgFactory.createElement("svg:g");

    g.setAttribute("stroke-width", borderStyle.width || 1);
    g.setAttribute("stroke-linecap", "round");
    g.setAttribute("stroke-linejoin", "round");

    for (let i = 0, ii = inkLists.length; i < ii; i++) {
      const polyline = this.svgFactory.createElement(this.svgElementName);
      polyline.setAttribute("points", inkLists[i].join(","));
      g.append(polyline);
    }

    this.container.append(svg);
    return this.container;
  }

  updateEdited(params) {
    // 更新批注显示
    const { thickness, points, rect } = params;
    if (thickness >= 0) {
      this.#polylinesGroupElement.setAttribute("stroke-width", thickness || 1);
    }
    if (points) {
      for (let i = 0, ii = this.#polylines.length; i < ii; i++) {
        this.#polylines[i].setAttribute("points", points[i].join(","));
      }
    }
  }
}
```

## 6. 批注工具检测机制

### 6.1 序列化与反序列化

**序列化（保存）：** `src/display/editor/ink.js` `serialize` 方法（第247-286行）

```javascript
serialize(isForCopying = false) {
  if (this.isEmpty()) {
    return null;
  }

  if (this.deleted) {
    return this.serializeDeleted();
  }

  const { lines, points } = this.serializeDraw(isForCopying);
  const {
    _drawingOptions: {
      stroke,
      "stroke-opacity": opacity,
      "stroke-width": thickness,
    },
  } = this;

  const serialized = Object.assign(super.serialize(isForCopying), {
    color: AnnotationEditor._colorManager.convert(stroke),
    opacity,
    thickness,
    paths: {
      lines,
      points,
    },
  });

  this.addComment(serialized);

  if (isForCopying) {
    serialized.isCopy = true;
    return serialized;
  }

  if (this.annotationElementId && !this.#hasElementChanged(serialized)) {
    return null;  // 没有变化，不需要更新
  }

  serialized.id = this.annotationElementId;
  return serialized;
}
```

**反序列化（加载）：** `src/display/editor/ink.js` `deserialize` 方法（第140-190行）

```javascript
static async deserialize(data, parent, uiManager) {
  let initialData = null;

  if (data instanceof InkAnnotationElement) {
    // 从 PDF 批注元素加载
    const {
      data: {
        inkLists,
        rect,
        rotation,
        id,
        color,
        opacity,
        borderStyle: { rawWidth: thickness },
        popupRef,
        richText,
        contentsObj,
      },
      parent: {
        page: { pageNumber },
      },
    } = data;

    initialData = data = {
      annotationType: AnnotationEditorType.INK,
      color: Array.from(color),
      thickness,
      opacity,
      paths: { points: inkLists },
      boxes: null,
      pageIndex: pageNumber - 1,
      rect: rect.slice(0),
      rotation,
      annotationElementId: id,
      id,
      deleted: false,
      popupRef,
      richText,
      comment: contentsObj?.str || null,
    };
  }

  const editor = await super.deserialize(data, parent, uiManager);
  editor._initialData = initialData;

  if (data.comment) {
    editor.setCommentData(data);
  }

  return editor;
}
```

### 6.2 批注检测流程

**启用编辑模式时：** `src/display/editor/annotation_editor_layer.js` `enable` 方法（第306-347行）

```javascript
async enable() {
  this.#isEnabling = true;

  const annotationElementIds = new Set();

  // 1. 显示现有编辑器
  for (const editor of this.#allEditorsIterator) {
    editor.enableEditing();
    editor.show(true);
    if (editor.annotationElementId) {
      this.#uiManager.removeChangedExistingAnnotation(editor);
      annotationElementIds.add(editor.annotationElementId);
    }
  }

  // 2. 从批注层获取可编辑的批注
  const annotationLayer = this.#annotationLayer;
  if (annotationLayer) {
    for (const editable of annotationLayer.getEditableAnnotations()) {
      // 隐藏原始批注元素
      editable.hide();

      if (this.#uiManager.isDeletedAnnotationElement(editable.data.id)) {
        continue;
      }

      if (annotationElementIds.has(editable.data.id)) {
        continue;
      }

      // 3. 反序列化批注为编辑器
      const editor = await this.deserialize(editable);
      if (!editor) {
        continue;
      }

      // 4. 添加到编辑器层
      this.addOrRebuild(editor);
      editor.enableEditing();
    }
  }

  this.#isEnabling = false;
}
```

**禁用编辑模式时：** `src/display/editor/annotation_editor_layer.js` `disable` 方法（第352-466行）

```javascript
disable() {
  this.#isDisabling = true;

  const annotationLayer = this.#annotationLayer;
  const needFakeAnnotation = [];

  if (annotationLayer) {
    const changedAnnotations = new Map();
    const resetAnnotations = new Map();

    // 1. 检查编辑器是否有变化
    for (const editor of this.#allEditorsIterator) {
      editor.disableEditing();

      if (!editor.annotationElementId) {
        needFakeAnnotation.push(editor);
        continue;
      }

      if (editor.serialize() !== null) {
        // 有变化，需要更新
        changedAnnotations.set(editor.annotationElementId, editor);
        continue;
      } else {
        // 无变化，重置
        resetAnnotations.set(editor.annotationElementId, editor);
      }

      this.getEditableAnnotation(editor.annotationElementId)?.show();
      editor.remove();
    }

    // 2. 更新批注元素
    const editables = annotationLayer.getEditableAnnotations();
    for (const editable of editables) {
      const { id } = editable.data;

      if (this.#uiManager.isDeletedAnnotationElement(id)) {
        editable.updateEdited({ deleted: true });
        continue;
      }

      let editor = resetAnnotations.get(id);
      if (editor) {
        editor.resetAnnotationElement(editable);
        editor.show(false);
        editable.show();
        continue;
      }

      editor = changedAnnotations.get(id);
      if (editor) {
        this.#uiManager.addChangedExistingAnnotation(editor);
        if (editor.renderAnnotationElement(editable)) {
          // 内容已改变，隐藏编辑器
          editor.show(false);
        }
      }

      editable.show();
    }
  }

  this.#isDisabling = false;
}
```

```
设置 CSS 类 .inkEditing
    ↓
启用点击事件监听 enableClick()
```

### 8.2 绘图流程

```
用户按下鼠标 (pointerdown)
    ↓
annotation_editor_layer.js 调用 startDrawingSession()
    ↓
DrawingEditor.startDrawing() 初始化绘图
    ↓
创建 InkDrawOutliner 实例
    ↓
在 DrawLayer 中创建 SVG 元素
    ↓
用户移动鼠标 (pointermove)
    ↓
DrawingEditor._drawMove() 处理移动
    ↓

    ↓

    ↓
用户释放鼠标 (pointerup)
    ↓

    ↓

    ↓

    ↓
创建 InkEditor 实例并添加到层
```

### 8.3 保存与加载流程

```

    ↓

    ↓

```

## 9. 关键技术点

### 9.1 贝塞尔曲线平滑

使用贝塞尔曲线来平滑连接用户绘制的点，使线条更自然：

```javascript
static createBezierPoints(x1, y1, x2, y2, x, y) {
  // 计算控制点
  const [c1x, c1y] = [(2 * x1 + x2) / 3, (2 * y1 + y2) / 3];
  const [c2x, c2y] = [(x1 + 2 * x2) / 3, (y1 + 2 * y2) / 3];
  return [c1x, c1y, c2x, c2y, x, y];
}
```

### 9.2 坐标归一化

将屏幕坐标转换为相对于页面的归一化坐标（0-1范围）：

```javascript
#normalizePoint(x, y) {
  return Outline._normalizePoint(
    x,
    y,
    this.#parentWidth,
    this.#parentHeight,
    this.#rotation
  );
}
```

### 9.3 多指触控处理

使用 `CurrentPointers` 类管理多指触控，支持手势识别：

```javascript
if (CurrentPointers.isUsingMultiplePointers()) {
  // 用户使用多个手指，结束绘图
  this._endDraw(event);
  return;
}
```

### 9.4 撤销/重做支持

通过命令模式实现撤销/重做：

```javascript
parent.addCommands({
  cmd: () => {
    parent.drawLayer.updateProperties(drawId, draw.setLastElement(lastElement));
  },
  undo: () => {
    parent.drawLayer.updateProperties(drawId, draw.removeLastElement());
  },
  mustExec: false,
  type: AnnotationEditorParamsType.DRAW_STEP,
});
```

### 9.5 旋转适配

支持页面旋转时的坐标转换：

```javascript
#rotateBox() {
  const { x, y, width, height, rotation, parentRotation } = this;
  switch ((rotation * 4 + parentRotation) / 90) {
    case 1: return [1 - y - height, x, height, width];
    case 2: return [1 - x - width, 1 - y - height, width, height];
    // ... 其他旋转角度
  }
}
```

## 10. 总结

PDF.js 的绘图工具实现是一个完善的系统，包括：

1. **工具切换**：通过事件系统实现模式切换，更新 UI 状态
2. **绘图机制**：使用 SVG 技术和贝塞尔曲线实现平滑绘图
3. **前端渲染**：DrawLayer 管理 SVG 元素，支持实时更新
4. **批注集成**：与 PDF 批注系统无缝对接，支持保存和加载
5. **参数配置**：提供颜色、粗细、不透明度等可配置参数
6. **撤销/重做**：完整的命令模式支持
7. **多点触控**：智能识别手势，支持触摸设备

这个设计既保证了用户体验的流畅性，又保持了与 PDF 标准的兼容性。
