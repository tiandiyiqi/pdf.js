```javascript
// 修改前
setFillRGBColor(opIdx, color) {
  this.dependencyTracker?.recordSimpleData("fillColor", opIdx);
  this.ctx.fillStyle = this.current.fillColor = color;  // ❌ 只接受RGB字符串
  this.current.patternFill = false;

// 修改后
setFillRGBColor(opIdx, color) {
  this.dependencyTracker?.recordSimpleData("fillColor", opIdx);

  // ✅ 处理ColorValue对象
  if (color instanceof ColorValue) {
    this.current.fillColorValue = color;  // ✅ 存储ColorValue
    this.ctx.fillStyle = this.current.fillColor = color.toRGB();  // ✅ 延迟转RGB
  } else {
    // 向后兼容：直接使用RGB字符串
    this.current.fillColorValue = null;
    this.ctx.fillStyle = this.current.fillColor = color;
  }

  this.current.patternFill = false;
}
```
