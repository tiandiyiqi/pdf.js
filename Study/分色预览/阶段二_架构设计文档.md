# PDF.js DeviceN颜色处理 - 阶段二架构设计文档

**项目**: PDF.js CMYK+多专色透明度混合改造  
**阶段**: 第二阶段 - 架构设计  
**日期**: 2025年12月10日  
**版本**: v1.0

## 执行摘要

本文档定义了支持DeviceN多通道颜色处理的完整架构方案。核心设计理念是：**在整个渲染管线中保持颜色通道的独立性，延迟RGB转换至最终显示阶段**。

### 设计原则

1. **通道独立性**: C、M、Y、K及各专色通道在处理过程中完全独立
2. **延迟转换**: RGB转换仅在最终显示时进行
3. **向后兼容**: RGB场景保持现有流程不变
4. **可扩展性**: 支持任意数量的专色通道
5. **性能优先**: 通过并行处理和优化算法确保性能

```javascript
/**
 * 多通道颜色值对象
 * 支持RGB、CMYK、DeviceN等多种颜色空间
 */
class ColorValue {
  /**
   * @param {Object} config
   * @param {string} config.colorSpace - 颜色空间类型: 'RGB', 'CMYK', 'DeviceN', 'Gray'
   * @param {Object} config.channels - 通道值
   * @param {Array<string>} config.channelNames - 通道名称 (DeviceN专用)
   * @param {string} config.rgbFallback - RGB备用值 (用于不支持的渲染器)
   */
  constructor(config) {
    this.colorSpace = config.colorSpace;
    this.channels = config.channels;
    this.channelNames = config.channelNames || null;
    this.rgbFallback = config.rgbFallback || null;
  }

  /**
   * 获取CMYK通道值
   * @returns {Array<number>} [C, M, Y, K] 或 null
   */
  getCMYK() {
    if (this.colorSpace === "CMYK") {
      return this.channels.cmyk;
    }
    if (this.colorSpace === "DeviceN" && this.channels.cmyk) {
      return this.channels.cmyk;
    }
    return null;
  }

  /**
   * 获取专色通道值
   * @param {string} spotName - 专色名称
   * @returns {number|null} 专色值 [0-1]
   */
  getSpot(spotName) {
    if (this.colorSpace === "DeviceN" && this.channels.spots) {
      return this.channels.spots[spotName] || null;
    }
    return null;
  }

  /**
   * 获取所有专色通道
   * @returns {Object} {spotName: value, ...}
   */
  getAllSpots() {
    if (this.colorSpace === "DeviceN" && this.channels.spots) {
      return { ...this.channels.spots };
    }
    return {};
  }

  /**
   * 转换为RGB (仅在必要时)
   * @returns {string} RGB hex值，如 "#RRGGBB"
   */
  toRGB() {
    if (this.rgbFallback) {
      return this.rgbFallback;
    }
    // 执行实际转换
    return this.#convertToRGB();
  }

  /**
   * 克隆颜色值对象
   * @returns {ColorValue}
   */
  clone() {
    return new ColorValue({
      colorSpace: this.colorSpace,
      channels: JSON.parse(JSON.stringify(this.channels)),
      channelNames: this.channelNames ? [...this.channelNames] : null,
      rgbFallback: this.rgbFallback,
    });
  }

  /**
   * 私有方法：执行RGB转换
   */
  #convertToRGB() {
    switch (this.colorSpace) {
      case "RGB":
        return this.channels.rgb;
      case "CMYK":
        return ColorConverter.cmykToRgb(this.channels.cmyk);
      case "DeviceN":
        // DeviceN先合并所有通道，再转RGB
        return ColorConverter.deviceNToRgb(this.channels);
      case "Gray":
        return ColorConverter.grayToRgb(this.channels.gray);
      default:
        return "#000000";
    }
  }
}

/**
 * 颜色值构建器
 */
class ColorValueBuilder {
  /**
   * 创建CMYK颜色值
   * @param {Array<number>} cmyk - [C, M, Y, K]
   * @returns {ColorValue}
   */
  static createCMYK(cmyk) {
    return new ColorValue({
      colorSpace: "CMYK",
      channels: { cmyk },
      rgbFallback: ColorConverter.cmykToRgb(cmyk),
    });
  }

  /**
   * 创建DeviceN颜色值
   * @param {Array<string>} channelNames - 通道名称列表
   * @param {Array<number>} values - 通道值列表
   * @returns {ColorValue}
   */
  static createDeviceN(channelNames, values) {
    // 分离CMYK和专色
    const cmyk = [];
    const spots = {};

    channelNames.forEach((name, i) => {
      if (["Cyan", "Magenta", "Yellow", "Black"].includes(name)) {
        cmyk.push(values[i]);
      } else {
        spots[name] = values[i];
      }
    });

    return new ColorValue({
      colorSpace: "DeviceN",
      channels: { cmyk, spots },
      channelNames,
      rgbFallback: ColorConverter.deviceNToRgb({ cmyk, spots }),
    });
  }

  /**
   * 创建RGB颜色值 (向后兼容)
   * @param {string} rgb - RGB hex值
   * @returns {ColorValue}
   */
  static createRGB(rgb) {
    return new ColorValue({
      colorSpace: "RGB",
      channels: { rgb },
      rgbFallback: rgb,
    });
  }
}

export { ColorValue, ColorValueBuilder };
```

```javascript
    if (colorValue.colorSpace === "RGB") {
      this.addOp(OPS.setStrokeRGBColor, colorValue.toRGB());
    } else if (colorValue.colorSpace === "CMYK") {
      this.addOp(OPS.setStrokeCMYKColor, colorValue);
    } else if (colorValue.colorSpace === "DeviceN") {
      this.addOp(OPS.setStrokeDeviceNColor, colorValue);
    } else {
      this.addOp(OPS.setStrokeColor, colorValue);
    }
  }

  /**
   * 序列化时处理ColorValue对象
   */
  #serializeColorValue(colorValue) {
    if (colorValue instanceof ColorValue) {
      return {
        __type__: "ColorValue",
        colorSpace: colorValue.colorSpace,
        channels: colorValue.channels,
        channelNames: colorValue.channelNames,
        rgbFallback: colorValue.rgbFallback,
      };
    }
    return colorValue;
  }

  /**
   * 反序列化ColorValue对象
   */
  static #deserializeColorValue(obj) {
    if (obj && obj.__type__ === "ColorValue") {
      return new ColorValue({
        colorSpace: obj.colorSpace,
        channels: obj.channels,
        channelNames: obj.channelNames,
        rgbFallback: obj.rgbFallback,
      });
    }
    return obj;
  }
}
```

```javascript
/**
 * DeviceN色彩空间管理器
 * 负责DeviceN色彩空间的解析、通道管理和混合空间确定
 */
class DeviceNManager {
  constructor() {
    /**
     * 通道管理器
     * @type {ChannelManager}
     */
    this.channelManager = new ChannelManager();

    /**
     * 混合颜色空间缓存
     * @type {Map<string, string>}
     */
    this.blendSpaceCache = new Map();
  }

  /**
   * 解析DeviceN色彩空间
   * @param {Array} colorSpaceArray - PDF色彩空间数组
   * @param {Object} xref - PDF交叉引用表
   * @returns {DeviceNColorSpace}
   */
  parseDeviceN(colorSpaceArray, xref) {
    // colorSpaceArray格式: [/DeviceN, names, alternateCS, tintFn, attributes]
    const names = xref.fetchIfRef(colorSpaceArray[1]);
    const alternateCS = colorSpaceArray[2];
    const tintFn = colorSpaceArray[3];
    const attributes = colorSpaceArray[4] || null;

    // 解析通道名称
    const channelNames = Array.isArray(names)
      ? names.map(n => n.name || n.toString())
      : [names.name || names.toString()];

    // 创建DeviceN色彩空间对象
    return new DeviceNColorSpace({
      channelNames,
      alternateCS,
      tintFn,
      attributes,
      numComps: channelNames.length,
    });
  }

  /**
   * 确定混合颜色空间
   * @param {Array<ColorValue>} colors - 参与混合的颜色列表
   * @returns {string} 混合颜色空间类型
   */
  determineBlendSpace(colors) {
    // 规则优先级：
    // 1. 如果有DeviceN颜色，使用DeviceN空间
    // 2. 如果有CMYK颜色，使用CMYK空间
    // 3. 否则使用RGB空间

    let hasDeviceN = false;
    let hasCMYK = false;

    for (const color of colors) {
      if (color.colorSpace === "DeviceN") {
        hasDeviceN = true;
        break;
      }
      if (color.colorSpace === "CMYK") {
        hasCMYK = true;
      }
    }

    if (hasDeviceN) return "DeviceN";
    if (hasCMYK) return "CMYK";
    return "RGB";
  }

  /**
   * 识别预览图层
   * @param {OperatorList} opList - 操作符列表
   * @returns {Array<PreviewLayer>} 预览图层列表
   */
  identifyPreviewLayers(opList) {
    const layers = [];
    let currentLayer = null;

    // 扫描操作符列表，识别图层标记
    // 预览图层通常有特定的命名模式或标记
    for (let i = 0; i < opList.fnArray.length; i++) {
      const fn = opList.fnArray[i];
      const args = opList.argsArray[i];

      // 检测图层开始标记 (如 /OC 可选内容)
      if (fn === OPS.beginMarkedContent) {
        currentLayer = this.#createLayer(args);
      }

      // 收集图层内的颜色信息
      if (currentLayer && this.#isColorOp(fn)) {
        currentLayer.addColorOp(fn, args);
      }

      // 检测图层结束
      if (fn === OPS.endMarkedContent && currentLayer) {
        layers.push(currentLayer);
        currentLayer = null;
      }
    }

    return layers;
  }

  /**
   * 创建预览图层对象
   */
  #createLayer(args) {
    return {
      name: args[0],
      colorOps: [],
      addColorOp(fn, args) {
        this.colorOps.push({ fn, args });
      },
    };
  }

  /**
   * 判断是否为颜色操作
   */
  #isColorOp(fn) {
    return [
      OPS.setFillRGBColor,
      OPS.setFillCMYKColor,
      OPS.setFillDeviceNColor,
      OPS.setStrokeRGBColor,
      OPS.setStrokeCMYKColor,
      OPS.setStrokeDeviceNColor,
    ].includes(fn);
  }
}

/**
 * DeviceN色彩空间类
 */
class DeviceNColorSpace {
  constructor(config) {
    this.channelNames = config.channelNames;
    this.alternateCS = config.alternateCS;
    this.tintFn = config.tintFn;
    this.attributes = config.attributes;
    this.numComps = config.numComps;

    // 分离CMYK通道和专色通道
    this.cmykIndices = [];
    this.spotIndices = [];

    const cmykNames = ["Cyan", "Magenta", "Yellow", "Black"];
    this.channelNames.forEach((name, i) => {
      if (cmykNames.includes(name)) {
        this.cmykIndices.push(i);
      } else {
        this.spotIndices.push(i);
      }
    });
  }

  /**
   * 从值数组创建ColorValue
   * @param {Array<number>} values - 通道值数组
   * @returns {ColorValue}
   */
  createColorValue(values) {
    const cmyk = this.cmykIndices.map(i => values[i]);
    const spots = {};

    this.spotIndices.forEach(i => {
      spots[this.channelNames[i]] = values[i];
    });

    return ColorValueBuilder.createDeviceN(this.channelNames, values);
  }

  /**
   * 获取专色通道名称列表
   * @returns {Array<string>}
   */
  getSpotNames() {
    return this.spotIndices.map(i => this.channelNames[i]);
  }
}

/**
 * 通道管理器
 */
class ChannelManager {
  constructor() {
    /**
     * 通道值存储
     * @type {Map<string, number>}
     */
    this.channels = new Map();
  }

  /**
   * 设置通道值
   * @param {string} channelName - 通道名称
   * @param {number} value - 通道值 [0-1]
   */
  setChannel(channelName, value) {
    this.channels.set(channelName, Math.max(0, Math.min(1, value)));
  }

  /**
   * 获取通道值
   * @param {string} channelName - 通道名称
   * @returns {number|null}
   */
  getChannel(channelName) {
    return this.channels.get(channelName) || null;
  }

  /**
   * 分离通道
   * 将DeviceN颜色拆分为各个独立通道
   * @param {ColorValue} colorValue
   * @returns {Map<string, number>}
   */
  separateChannels(colorValue) {
    const separated = new Map();

    if (colorValue.colorSpace === "DeviceN") {
      // CMYK通道
      const cmykNames = ["Cyan", "Magenta", "Yellow", "Black"];
      colorValue.getCMYK().forEach((val, i) => {
        separated.set(cmykNames[i], val);
      });

      // 专色通道
      Object.entries(colorValue.getAllSpots()).forEach(([name, val]) => {
        separated.set(name, val);
      });
    }

    return separated;
  }

  /**
   * 合并通道
   * 将独立通道合并为DeviceN颜色
   * @param {Map<string, number>} channels - 通道值映射
   * @param {Array<string>} channelNames - 通道名称顺序
   * @returns {ColorValue}
   */
  mergeChannels(channels, channelNames) {
    const values = channelNames.map(name => channels.get(name) || 0);
    return ColorValueBuilder.createDeviceN(channelNames, values);
  }

  /**
   * 重置所有通道
   */
  reset() {
    this.channels.clear();
  }
}

export { DeviceNManager, DeviceNColorSpace, ChannelManager };
```

```javascript
/**
 * 颜色转换器
 * 提供各种颜色空间之间的转换
 */
class ColorConverter {
  /**
   * CMYK转RGB
   * @param {Array<number>} cmyk - [C, M, Y, K]
   * @returns {string} RGB hex值
   */
  static cmykToRgb(cmyk) {
    const [c, m, y, k] = cmyk;

    // 标准CMYK→RGB转换公式
    const r = 255 * (1 - c) * (1 - k);
    const g = 255 * (1 - m) * (1 - k);
    const b = 255 * (1 - y) * (1 - k);

    return this.#rgbToHex(r, g, b);
  }

  /**
   * RGB转CMYK
   * @param {string} rgb - RGB hex值
   * @returns {Array<number>} [C, M, Y, K]
   */
  static rgbToCmyk(rgb) {
    const { r, g, b } = this.#hexToRgb(rgb);

    // 标准RGB→CMYK转换公式
    const r1 = r / 255;
    const g1 = g / 255;
    const b1 = b / 255;

    const k = 1 - Math.max(r1, g1, b1);

    if (k === 1) {
      return [0, 0, 0, 1];
    }

    const c = (1 - r1 - k) / (1 - k);
    const m = (1 - g1 - k) / (1 - k);
    const y = (1 - b1 - k) / (1 - k);

    return [c, m, y, k];
  }

  /**
   * DeviceN转RGB
   * @param {Object} channels - {cmyk: [...], spots: {...}}
   * @returns {string} RGB hex值
   */
  static deviceNToRgb(channels) {
    // 首先将所有通道合并到CMYK空间
    const cmyk = [...channels.cmyk];

    // 专色通道叠加到CMYK (简化处理，实际应使用tint函数)
    Object.values(channels.spots).forEach(spotValue => {
      // 专色通常叠加到K通道 (这是简化处理)
      cmyk[3] = Math.min(1, cmyk[3] + spotValue * 0.3);
    });

    return this.cmykToRgb(cmyk);
  }

  /**
   * 灰度转RGB
   * @param {number} gray - 灰度值 [0-1]
   * @returns {string} RGB hex值
   */
  static grayToRgb(gray) {
    const val = Math.round(gray * 255);
    return this.#rgbToHex(val, val, val);
  }

  /**
   * RGB转Hex
   */
  static #rgbToHex(r, g, b) {
    r = Math.round(Math.max(0, Math.min(255, r)));
    g = Math.round(Math.max(0, Math.min(255, g)));
    b = Math.round(Math.max(0, Math.min(255, b)));

    return `#${r.toString(16).padStart(2, "0")}${g.toString(16).padStart(2, "0")}${b.toString(16).padStart(2, "0")}`;
  }

  /**
   * Hex转RGB
   */
  static #hexToRgb(hex) {
    const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    return result
      ? {
          r: parseInt(result[1], 16),
          g: parseInt(result[2], 16),
          b: parseInt(result[3], 16),
        }
      : { r: 0, g: 0, b: 0 };
  }
}

export { ColorConverter };
```

```javascript
  blend(backdrop, source, alpha) {
    // 分离通道
    const channelManager = new ChannelManager();
    const backChannels = channelManager.separateChannels(backdrop);
    const srcChannels = channelManager.separateChannels(source);

    // 获取所有通道名称
    const allChannelNames = new Set([
      ...backChannels.keys(),
      ...srcChannels.keys(),
    ]);

    // 对每个通道独立混合
    const resultChannels = new Map();
    for (const channelName of allChannelNames) {
      const backVal = backChannels.get(channelName) || 0;
      const srcVal = srcChannels.get(channelName) || 0;
      const blendedVal = this.blendChannel(backVal, srcVal, alpha);
      resultChannels.set(channelName, blendedVal);
    }

    // 合并通道
    return channelManager.mergeChannels(
      resultChannels,
      Array.from(allChannelNames)
    );
  }
```
