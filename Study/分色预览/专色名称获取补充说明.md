# 专色名称获取补充说明

## 专色名称获取方法

在PDF.js中，专色名称主要通过以下几种方式获取：

### 1. 从DeviceN色彩空间解析

在 `src/core/evaluator.js` 中，当解析到DeviceN或Separation色彩空间时，可以从色彩空间定义中获取专色名称：

```javascript
// evaluator.js 中的相关代码
case OPS.setFillColorN:
case OPS.setStrokeColorN:
  const cs = stateManager.state.fillColorSpace;
  if (cs.name === "DeviceN" || cs.name === "Separation") {
    // 从色彩空间获取通道名称
    const channelNames = cs.channelNames || [];

    // 分离CMYK和专色名称
    const cmykNames = ["Cyan", "Magenta", "Yellow", "Black"];
    const spotNames = channelNames.filter(name => !cmykNames.includes(name));

    console.log("发现的专色名称:", spotNames);
  }
  break;
```

### 2. 从ColorValue对象获取

通过ColorValue的 `getAllSpots()` 方法获取所有专色名称：

```javascript
// 在渲染过程中获取专色名称
if (colorValue.colorSpace === "DeviceN") {
  const spotChannels = colorValue.getAllSpots();
  const spotNames = Object.keys(spotChannels);

  // 将这些专色名称添加到过滤器配置中
  spotNames.forEach(spotName => {
    ColorConverter.addSpotColor(spotName, true);
  });
}
```

### 3. 使用DeviceNManager自动发现

创建自动发现专色的工具函数：

```javascript
// 专色自动发现工具
class SpotColorDiscoverer {
  constructor() {
    this.discoveredSpots = new Set();
  }

  /**
   * 从操作符列表中自动发现专色
   * @param {Object} opList - PDF操作符列表
   */
  discoverFromOpList(opList) {
    if (!opList?.fnArray || !opList?.argsArray) return;

    for (let i = 0; i < opList.fnArray.length; i++) {
      const fn = opList.fnArray[i];
      const args = opList.argsArray[i];

      // 检查颜色设置操作
      if (this.#isColorOperation(fn) && args?.[0] instanceof ColorValue) {
        const colorValue = args[0];
        this.#extractSpotNames(colorValue);
      }
    }
  }

  /**
   * 从ColorValue中提取专色名称
   * @param {ColorValue} colorValue
   */
  #extractSpotNames(colorValue) {
    if (colorValue.colorSpace === "DeviceN") {
      const spots = colorValue.getAllSpots();
      Object.keys(spots).forEach(spotName => {
        this.discoveredSpots.add(spotName);
      });
    }
  }

  /**
   * 判断是否为颜色操作
   */
  #isColorOperation(fn) {
    const colorOps = [
      OPS.setFillRGBColor,
      OPS.setFillCMYKColor,
      OPS.setStrokeRGBColor,
      OPS.setStrokeCMYKColor,
      OPS.setFillColorN,
      OPS.setStrokeColorN,
    ];
    return colorOps.includes(fn);
  }

  /**
   * 获取所有发现的专色名称
   */
  getDiscoveredSpots() {
    return Array.from(this.discoveredSpots);
  }

  /**
   * 将发现的专色应用到过滤器
   */
  applyToFilter() {
    this.discoveredSpots.forEach(spotName => {
      ColorConverter.addSpotColor(spotName, true);
    });
  }
}
```

## 初始化流程建议

### 方案A：预扫描PDF获取专色名称

```javascript
// 在PDF加载完成后预扫描专色
async function initializeColorFilter(pdfDocument) {
  const discoverer = new SpotColorDiscoverer();

  // 扫描所有页面
  for (let i = 1; i <= pdfDocument.numPages; i++) {
    const page = await pdfDocument.getPage(i);
    const opList = await page.getOperatorList();

    discoverer.discoverFromOpList(opList);
  }

  // 获取发现的专色
  const spotNames = discoverer.getDiscoveredSpots();

  // 构建颜色配置
  const colorConfig = {
    enabled: true,
    colors: {
      Cyan: true,
      Magenta: true,
      Yellow: true,
      Black: true,
    },
  };

  // 添加发现的专色
  spotNames.forEach(spotName => {
    colorConfig.colors[spotName] = true;
  });

  // 设置过滤器
  ColorConverter.setColorFilterConfig(colorConfig);

  console.log(`初始化完成，发现 ${spotNames.length} 个专色:`, spotNames);
}
```

### 方案B：运行时动态添加

```javascript
// 在颜色转换时动态添加专色
static deviceNToRgbWithFilter(channels) {
  // 先处理专色名称的自动注册
  if (channels.spots) {
    Object.keys(channels.spots).forEach(spotName => {
      // 如果专色尚未在配置中，自动添加
      if (!this.#colorFilterConfig.colors.has(spotName)) {
        this.#colorFilterConfig.colors.set(spotName, true);
        console.log(`自动注册专色: ${spotName}`);
      }
    });
  }

  // 后续过滤逻辑保持不变
  const cmyk = channels.cmyk ? this.#filterCMYK(channels.cmyk) : [0, 0, 0, 0];

  let totalSpot = 0;
  if (channels.spots) {
    for (const [spotName, spotValue] of Object.entries(channels.spots)) {
      totalSpot += this.#filterSpot(spotName, spotValue);
    }
  }

  cmyk[3] = Math.min(1, cmyk[3] + totalSpot * 0.3);
  return this.cmykToRgb(cmyk);
}
```

## 实际使用建议

推荐使用**方案B（运行时动态添加）**，因为：

1. **无需预扫描**：避免性能开销
2. **按需注册**：只在遇到专色时才添加到配置中
3. **自动完成**：用户无需手动配置专色名称
4. **兼容性好**：适用于各种PDF文件

只需在初始化时设置CMYK基础配置，专色会在渲染过程中自动发现和注册。
