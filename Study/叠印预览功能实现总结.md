// 叠印选项初始化
initializeOverprintOption() {
    const overprintOptionCheckbox = document.getElementById("overprintOption");
    const overprintOptionButton = document.getElementById("overprintOptionButton");

    const initialValue = AppOptions.get("overprintOption");
    overprintOptionCheckbox.checked = initialValue;
    this.updateOverprintButton(overprintOptionButton, initialValue);

    // 按钮点击事件
    overprintOptionButton.addEventListener("click", () => {
        const newValue = !overprintOptionCheckbox.checked;
        overprintOptionCheckbox.checked = newValue;
        this.setOverprintOption(newValue);
    });
}

// 设置叠印选项
setOverprintOption(enabled) {
    AppOptions.set("overprintOption", enabled);
    // 重新渲染当前页面
    if (this.pdfViewer) {
        // 获取当前可见页面
        const visiblePages = this.pdfViewer._getVisiblePages();
        // 重置可见页面的渲染状态，强制重新渲染
        for (const { view } of visiblePages.views) {
            if (view.renderingState === RenderingStates.FINISHED) {
                view.reset({ keepCanvasWrapper: true });
            }
        }
        // 触发重新渲染
        this.pdfViewer.forceRendering();
    }
}

// 处理父页面消息
handleParentMessage(data) {
    if (data.type === "setOverprintOption") {
        this.setOverprintOption(data.enabled);
    }
}

1. **渲染状态检查问题**：`pdf_rendering_queue.js` 中的 `renderView` 方法在页面状态为 `FINISHED` 时直接返回，不会重新渲染

3. **Canvas混合模式实现正确**：CanvasGraphics中正确接收并应用了`overprintOption`参数 ✅

**解决方案：**

在 `setOverprintOption` 方法中，先重置已渲染完成的页面状态为 `INITIAL`，然后再触发重新渲染。这样确保页面能够响应叠印选项的变化。

## 5. 修复实施

### 5.1 已实施的修复方案

**修复文件：** `/web/app.js`

**修复内容：**

修改 `setOverprintOption` 方法，在触发重新渲染前先重置可见页面的渲染状态：

```javascript
setOverprintOption(enabled) {
    AppOptions.set("overprintOption", enabled);
    if (this.pdfViewer) {
        // 获取当前可见页面
        const visiblePages = this.pdfViewer._getVisiblePages();
        // 重置可见页面的渲染状态，强制重新渲染
        for (const { view } of visiblePages.views) {
            if (view.renderingState === RenderingStates.FINISHED) {
                view.reset({ keepCanvasWrapper: true });
            }
        }
        // 触发重新渲染
        this.pdfViewer.forceRendering();
    }
}
```

**修复原理：**

- `pdf_rendering_queue.js` 中的 `renderView` 方法在页面状态为 `FINISHED` 时不会重新渲染
- 通过调用 `view.reset({ keepCanvasWrapper: true })` 将页面状态重置为 `INITIAL`
- 然后调用 `forceRendering()` 触发重新渲染，此时页面会使用新的叠印选项重新绘制

**修复效果：**

- ✅ 点击工具栏中的"叠印预览"按钮可以正确切换叠印效果
- ✅ 页面能够实时响应叠印选项的变化
- ✅ 叠印效果在PDF内容中正确显示（CMY颜色叠加显示为黑色）
